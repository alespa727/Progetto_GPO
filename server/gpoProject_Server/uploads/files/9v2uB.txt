CREATE TABLE `users` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `username` varchar(20) UNIQUE NOT NULL,
  `password` varchar(256) NOT NULL,
  `isAdmin` boolean DEFAULT false,
  `timeLastAccess` datetime DEFAULT (CURRENT_TIMESTAMP),
  `createdAt` date DEFAULT (CURDATE),
  `imagePath` varchar(100)
);

CREATE TABLE `friendships` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkUser1` int NOT NULL,
  `fkUser2` int NOT NULL,
  `createdAt` date DEFAULT (CURDATE)
);

CREATE TABLE `chats` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkFriendship` int UNIQUE NOT NULL,
  `timeLastMessage` datetime,
  `createdAt` date DEFAULT (CURDATE)
);

CREATE TABLE `messagesChat` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkChat` int NOT NULL,
  `fkUser` int NOT NULL,
  `message` text,
  `sentAt` datetime DEFAULT (CURRENT_TIMESTAMP)
);

CREATE TABLE `attachments` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkMessage` int NOT NULL,
  `path` varchar(255)
);

CREATE TABLE `calls` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkChat` int NOT NULL,
  `startTime` datetime DEFAULT (CURRENT_TIMESTAMP),
  `endTime` datetime
);

CREATE TABLE `communities` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkUserOwner` int NOT NULL,
  `inviteCode` varchar(20) UNIQUE NOT NULL,
  `isInviteCodeValid` boolean DEFAULT false,
  `name` varchar(100) NOT NULL,
  `description` text,
  `createdAt` date DEFAULT (CURDATE)
);

CREATE TABLE `sections` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkCommunity` int NOT NULL,
  `name` varchar(30) NOT NULL
);

CREATE TABLE `channels` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkSection` int NOT NULL,
  `name` varchar(100) NOT NULL,
  `type` enum(testo,vocale) NOT NULL,
  `description` text,
  `createdAt` date DEFAULT (CURDATE)
);

CREATE TABLE `messagesCommunity` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `fkChannel` int NOT NULL,
  `fkUser` int NOT NULL,
  `message` text,
  `sentAt` datetime DEFAULT (CURRENT_TIMESTAMP)
);

CREATE TABLE `registrations` (
  `fkCommunity` int NOT NULL,
  `fkUser` int NOT NULL,
  `date` datetime DEFAULT (CURRENT_TIMESTAMP),
  PRIMARY KEY (`fkCommunity`, `fkUser`)
);

CREATE UNIQUE INDEX `friendships_index_0` ON `friendships` (`fkUser1`, `fkUser2`);
