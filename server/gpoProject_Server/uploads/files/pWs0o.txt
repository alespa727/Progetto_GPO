CREATE TABLE `utenti` (
  `id` int AUTO_INCREMENT,
  `username` varchar(20) PRIMARY KEY,
  `password` varchar(256),
  `is_admin` boolean,
  `ultimo_accesso` datetime,
  `created_at` timestamp,
  `image_path` varchar(100)
);

CREATE TABLE `chat` (
  `id` int PRIMARY KEY,
  `amicizia_fk` int,
  `created_at` timestamp,
  `last_message` datetime
);

CREATE TABLE `amicizie` (
  `id` int PRIMARY KEY,
  `username_fk1` varchar(20),
  `username_fk2` varchar(20),
  `created_at` timestamp
);

CREATE TABLE `chiamate` (
  `id` int PRIMARY KEY,
  `chat_fk` int,
  `start_time` datetime,
  `end_time` datetime,
  `is_finished` boolean
);

CREATE TABLE `community` (
  `id` int PRIMARY KEY,
  `owner_fk` varchar(20),
  `invite_code` varchar(20),
  `is_code_valid` boolean,
  `name` varchar(100),
  `description` text
);

CREATE TABLE `iscrizione` (
  `id` int PRIMARY KEY,
  `community_fk` int,
  `username_fk` varchar(20),
  `date` datetime
);

CREATE TABLE `sezioni_canali` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `nome` varchar(30),
  `community_fk` int
);

CREATE TABLE `canali` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `sezione_fk` int,
  `nome` varchar(100) NOT NULL,
  `tipo` varchar(20) NOT NULL COMMENT 'testo | vocale',
  `descrizione` text,
  `creato_il` datetime DEFAULT (CURRENT_TIMESTAMP)
);

CREATE TABLE `messaggi_community` (
  `id` int PRIMARY KEY AUTO_INCREMENT,
  `canale_fk` int,
  `username_fk` varchar(20),
  `text` text,
  `sent_at` timestamp
);

CREATE TABLE `messaggi_chat` (
  `id` int PRIMARY KEY,
  `chat_fk` int,
  `username_fk` varchar(20),
  `text` text,
  `sent_at` timestamp
);

CREATE TABLE `allegati` (
  `id` int PRIMARY KEY,
  `messaggio_chat_fk` int,
  `path` varchar(100)
);
